# 后缀数组

```cpp
struct SaArray{
	SaArray()=delete;
	SaArray(const char*st):S(st){}
	SaArray(const std::string&st):S(st){}
	std::string S;
	int n,m;
	std::vector<int>Sa,Id,Rk,Lc;
	void getSa(){
		n=S.size()-1;m=128;
		Sa.resize(n+10);
		Id.resize(n+10);
		Rk.resize(n+10);
		Lc.resize(n+10);
		int p;
		std::vector<int>Cnt(2100000);
		std::vector<int>OlRk(2*n+10);
		for(int i=1;i<=n;i++)Cnt[Rk[i]=S[i]]++;
		for(int i=1;i<=m;i++)Cnt[i]+=Cnt[i-1];
		for(int i=n;i>=1;i--)Sa[Cnt[Rk[i]]--]=i;
		for(int i=1;i<=m;i++)Cnt[i]=0;
		for(int w=1;w<=n;w<<=1,m=p){
			int cur=0;
			for(int i=n-w+1;i<=n;i++)Id[++cur]=i;//这些第二关键字都是0
			for(int i=1;i<=n;i++)
				if(Sa[i]>w)Id[++cur]=Sa[i]-w;//第二关键字排完序将自身的第一关键字扔进队列
			for(int i=1;i<=n;i++)Cnt[Rk[Id[i]]]++;
			for(int i=1;i<=m;i++)Cnt[i]+=Cnt[i-1];
			for(int i=n;i>=1;i--)Sa[Cnt[Rk[Id[i]]]--]=Id[i];
			for(int i=1;i<=m;i++)Cnt[i]=0;
			p=0;
			for(int i=1;i<=n;i++)
				OlRk[i]=Rk[i];
			for(int i=1;i<=n;i++){
				if(OlRk[Sa[i]]==OlRk[Sa[i-1]]&&OlRk[Sa[i]+w]==OlRk[Sa[i-1]+w])
					Rk[Sa[i]]=p;
				else Rk[Sa[i]]=++p;
			}
			if(p==n)break;
		}
		for(int i=1,k=0;i<=n;i++){//Lc[Rk[i]]>=Lc[Rk[i-1]]-1
			if(Rk[i]==0)continue;
			if(k)--k;
			while(S[i+k]==S[Sa[Rk[i]-1]+k])++k;
			Lc[Rk[i]]=k;
		}
		//Lcp(Sa[i],Sa[j])=min_{i+1~j}Lc[k]
	}
};
```
