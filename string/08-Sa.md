# 后缀数组与 LCP

返回所有后缀起点的字典序排列，点编号为 0-based。实现会追加值为 0 的哨兵，因此输入字符串不能包含空字符。

```cpp
#include <bits/stdc++.h>
using namespace std;

pair<vector<int>, vector<int>> suffix_array_and_lcp(const string& text) {
    string extended = text;
    extended.push_back(char(0));
    int size = (int)extended.size();

    vector<int> suffix_array(size);
    vector<int> rank(size);
    {
        vector<int> count(max(256, size));
        for (unsigned char character : extended) ++count[character];
        for (int value = 1; value < (int)count.size(); ++value) {
            count[value] += count[value - 1];
        }
        for (int index = size - 1; index >= 0; --index) {
            unsigned char character = extended[index];
            suffix_array[--count[character]] = index;
        }
        int class_count = 1;
        for (int position = 1; position < size; ++position) {
            if (extended[suffix_array[position]] !=
                extended[suffix_array[position - 1]]) {
                ++class_count;
            }
            rank[suffix_array[position]] = class_count - 1;
        }
    }

    vector<int> shifted(size);
    vector<int> next_rank(size);
    for (int length = 1; length < size; length <<= 1) {
        for (int position = 0; position < size; ++position) {
            shifted[position] = suffix_array[position] - length;
            if (shifted[position] < 0) shifted[position] += size;
        }
        int class_count = *max_element(rank.begin(), rank.end()) + 1;
        vector<int> count(class_count);
        for (int index : shifted) ++count[rank[index]];
        for (int value = 1; value < class_count; ++value) {
            count[value] += count[value - 1];
        }
        for (int position = size - 1; position >= 0; --position) {
            int index = shifted[position];
            suffix_array[--count[rank[index]]] = index;
        }

        next_rank[suffix_array[0]] = 0;
        class_count = 1;
        for (int position = 1; position < size; ++position) {
            int current = suffix_array[position];
            int previous = suffix_array[position - 1];
            pair<int, int> current_pair{
                rank[current], rank[(current + length) % size]};
            pair<int, int> previous_pair{
                rank[previous], rank[(previous + length) % size]};
            if (current_pair != previous_pair) ++class_count;
            next_rank[current] = class_count - 1;
        }
        rank.swap(next_rank);
        if (class_count == size) break;
    }

    suffix_array.erase(suffix_array.begin());
    int text_size = (int)text.size();
    vector<int> suffix_rank(text_size);
    for (int position = 0; position < text_size; ++position) {
        suffix_rank[suffix_array[position]] = position;
    }

    vector<int> lcp(text_size);
    int matched = 0;
    for (int index = 0; index < text_size; ++index) {
        int position = suffix_rank[index];
        if (position == 0) continue;
        int previous = suffix_array[position - 1];
        while (index + matched < text_size &&
               previous + matched < text_size &&
               text[index + matched] == text[previous + matched]) {
            ++matched;
        }
        lcp[position] = matched;
        if (matched > 0) --matched;
    }
    return {suffix_array, lcp};
}
```

`lcp[i]` 是 `suffix_array[i-1]` 与 `suffix_array[i]` 所指后缀的最长公共前缀，`lcp[0]=0`。构造复杂度为 $O(n\log n)$，空间复杂度为 $O(n)$；任意两个后缀的 LCP 可转化为 `lcp` 区间最小值查询。
