# 左偏树

可合并优先队列。每个 `push` 返回一个新堆根，`merge` 会破坏性地合并两个根。默认是小根堆。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

template<class Value, class Compare = less<Value>>
class LeftistHeap {
    struct Node {
        Value value{};
        int child[2]{0, 0};
        int null_path_length = 0;
    };

    vector<Node> nodes{Node{}};
    Compare compare;

public:
    int merge(int lhs, int rhs) {
        if (lhs == 0 || rhs == 0) return lhs | rhs;
        if (compare(nodes[rhs].value, nodes[lhs].value)) swap(lhs, rhs);
        nodes[lhs].child[1] = merge(nodes[lhs].child[1], rhs);
        int left = nodes[lhs].child[0];
        int right = nodes[lhs].child[1];
        if (nodes[left].null_path_length < nodes[right].null_path_length) {
            swap(nodes[lhs].child[0], nodes[lhs].child[1]);
        }
        nodes[lhs].null_path_length =
            nodes[nodes[lhs].child[1]].null_path_length + 1;
        return lhs;
    }

    int push(int root, const Value& value) {
        int created = (int)nodes.size();
        nodes.push_back(Node{value, {0, 0}, 1});
        return merge(root, created);
    }

    const Value& top(int root) const {
        assert(root != 0);
        return nodes[root].value;
    }

    int pop(int root) {
        assert(root != 0);
        return merge(nodes[root].child[0], nodes[root].child[1]);
    }

    bool empty(int root) const {
        return root == 0;
    }
};
```

合并、插入和删除堆顶的最坏复杂度均为 $O(\log n)$。合并后旧根不应再作为独立堆使用。
