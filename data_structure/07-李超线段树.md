# 李超线段树

在整数横坐标区间上维护若干线段，并查询某个横坐标处的最大值。若函数值相同，返回编号较小的线段。直线和端点使用 `long double`；如果题目中的 $kx+b$ 能安全装入 `__int128`，可改用整数获得精确比较。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

class LiChaoSegmentTree {
    using ld = long double;
    static constexpr ld eps = 1e-12L;

    struct Line {
        ld slope;
        ld intercept;
        int id;

        ld value(ld x) const {
            return slope * x + intercept;
        }
    };

    int left_bound;
    int right_bound;
    vector<int> tree;
    vector<Line> lines;

    bool better(int lhs_id, int rhs_id, int x) const {
        if (lhs_id == -1) return false;
        if (rhs_id == -1) return true;
        ld difference = lines[lhs_id].value(x) - lines[rhs_id].value(x);
        if (fabsl(difference) > eps) return difference > 0;
        return lines[lhs_id].id < lines[rhs_id].id;
    }

    void insert_line(int node, int left, int right, int line_id) {
        if (tree[node] == -1) {
            tree[node] = line_id;
            return;
        }
        int middle = left + (right - left) / 2;
        if (better(line_id, tree[node], middle)) {
            swap(line_id, tree[node]);
        }
        if (left == right) return;
        if (better(line_id, tree[node], left)) {
            insert_line(node * 2, left, middle, line_id);
        } else if (better(line_id, tree[node], right)) {
            insert_line(node * 2 + 1, middle + 1, right, line_id);
        }
    }

    void insert_segment(int node, int left, int right,
                        int query_left, int query_right, int line_id) {
        if (query_left <= left && right <= query_right) {
            insert_line(node, left, right, line_id);
            return;
        }
        int middle = left + (right - left) / 2;
        if (query_left <= middle) {
            insert_segment(node * 2, left, middle,
                           query_left, query_right, line_id);
        }
        if (query_right > middle) {
            insert_segment(node * 2 + 1, middle + 1, right,
                           query_left, query_right, line_id);
        }
    }

public:
    LiChaoSegmentTree(int minimum_x, int maximum_x)
        : left_bound(minimum_x), right_bound(maximum_x),
          tree(4 * (maximum_x - minimum_x + 1), -1) {
        assert(minimum_x <= maximum_x);
    }

    int add_segment(ld slope, ld intercept,
                    int segment_left, int segment_right) {
        assert(left_bound <= segment_left && segment_left <= segment_right);
        assert(segment_right <= right_bound);
        int line_id = (int)lines.size();
        lines.push_back({slope, intercept, line_id});
        insert_segment(1, left_bound, right_bound,
                       segment_left, segment_right, line_id);
        return line_id;
    }

    optional<pair<ld, int>> query(int x) const {
        assert(left_bound <= x && x <= right_bound);
        int node = 1;
        int left = left_bound;
        int right = right_bound;
        int best_id = -1;
        while (true) {
            if (better(tree[node], best_id, x)) best_id = tree[node];
            if (left == right) break;
            int middle = left + (right - left) / 2;
            if (x <= middle) {
                node *= 2;
                right = middle;
            } else {
                node = node * 2 + 1;
                left = middle + 1;
            }
        }
        if (best_id == -1) return nullopt;
        return pair<ld, int>{lines[best_id].value(x), lines[best_id].id};
    }
};
```

插入一条覆盖整个值域的直线是 $O(\log C)$，插入线段是 $O(\log^2 C)$，单点查询是 $O(\log C)$，其中 $C$ 是离散整数横坐标范围长度。若横坐标很大但只查询预先给出的若干位置，应先离散化，并在线段端点处用原横坐标求值。
