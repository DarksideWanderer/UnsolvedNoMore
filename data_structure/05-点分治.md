# 点分治：树上距离存在性

给定非负边权树和若干询问，判断是否存在**两个不同点**的距离等于询问值。点分治在每个重心处只统计路径经过该重心的点对；同一子树中的距离必须先查询、后加入，避免重复计数。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

class CentroidDistanceQueries {
    using i64 = long long;
    struct Edge {
        int to;
        i64 weight;
    };

    int node_count;
    i64 maximum_query;
    vector<vector<Edge>> graph;
    vector<int> subtree_size;
    vector<bool> removed;
    vector<i64> queries;
    vector<bool> answer;

    int calculate_size(int node, int parent) {
        subtree_size[node] = 1;
        for (const Edge& edge : graph[node]) {
            if (edge.to != parent && !removed[edge.to]) {
                subtree_size[node] += calculate_size(edge.to, node);
            }
        }
        return subtree_size[node];
    }

    int find_centroid(int node, int parent, int component_size) {
        for (const Edge& edge : graph[node]) {
            if (edge.to != parent && !removed[edge.to] &&
                subtree_size[edge.to] > component_size / 2) {
                return find_centroid(edge.to, node, component_size);
            }
        }
        return node;
    }

    void collect_distances(int node, int parent, i64 distance,
                           vector<i64>& distances) const {
        if (distance > maximum_query) return;
        distances.push_back(distance);
        for (const Edge& edge : graph[node]) {
            if (edge.to != parent && !removed[edge.to]) {
                collect_distances(edge.to, node, distance + edge.weight,
                                  distances);
            }
        }
    }

    void process_centroid(int centroid) {
        unordered_set<i64> previous_distances;
        previous_distances.insert(0);
        for (const Edge& edge : graph[centroid]) {
            if (removed[edge.to]) continue;
            vector<i64> distances;
            collect_distances(edge.to, centroid, edge.weight, distances);
            for (i64 distance : distances) {
                for (int query_id = 0;
                     query_id < (int)queries.size(); ++query_id) {
                    if (!answer[query_id] && queries[query_id] >= distance &&
                        previous_distances.contains(
                            queries[query_id] - distance)) {
                        answer[query_id] = true;
                    }
                }
            }
            previous_distances.insert(distances.begin(), distances.end());
        }
    }

    void decompose(int start) {
        int component_size = calculate_size(start, -1);
        int centroid = find_centroid(start, -1, component_size);
        removed[centroid] = true;
        process_centroid(centroid);
        for (const Edge& edge : graph[centroid]) {
            if (!removed[edge.to]) decompose(edge.to);
        }
    }

public:
    explicit CentroidDistanceQueries(int initial_node_count)
        : node_count(initial_node_count), graph(initial_node_count),
          subtree_size(initial_node_count), removed(initial_node_count) {}

    void add_edge(int lhs, int rhs, i64 weight) {
        assert(weight >= 0);
        graph[lhs].push_back({rhs, weight});
        graph[rhs].push_back({lhs, weight});
    }

    vector<bool> solve(vector<i64> query_values) {
        queries = move(query_values);
        answer.assign(queries.size(), false);
        fill(removed.begin(), removed.end(), false);
        maximum_query =
            queries.empty() ? -1 : *max_element(queries.begin(), queries.end());
        if (node_count > 0 && maximum_query >= 0) decompose(0);
        return answer;
    }
};
```

设询问数为 $q$，复杂度为 $O(nq\log n)$，适合 $q$ 较小的版本。所有边权非负是剪枝 `distance > maximum_query` 正确的前提；负边权树不能使用该剪枝。
