# 动态开点线段树合并

维护固定整数值域上的权值和。根编号 0 表示空树；`merge` 破坏性合并两个根，合并后不能继续把右根当作独立版本使用。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

class MergeableSegmentTree {
    using i64 = long long;
    struct Node {
        int child[2]{0, 0};
        i64 sum = 0;
    };

    int left_bound;
    int right_bound;
    vector<Node> nodes{Node{}};

    int create() {
        nodes.push_back(Node{});
        return (int)nodes.size() - 1;
    }

    int update(int root, int left, int right, int position, i64 delta) {
        if (root == 0) root = create();
        nodes[root].sum += delta;
        if (left == right) return root;
        int middle = left + (right - left) / 2;
        if (position <= middle) {
            nodes[root].child[0] = update(
                nodes[root].child[0], left, middle, position, delta);
        } else {
            nodes[root].child[1] = update(
                nodes[root].child[1], middle + 1, right, position, delta);
        }
        return root;
    }

    int merge_roots(int lhs, int rhs, int left, int right) {
        if (lhs == 0 || rhs == 0) return lhs | rhs;
        nodes[lhs].sum += nodes[rhs].sum;
        if (left != right) {
            int middle = left + (right - left) / 2;
            nodes[lhs].child[0] = merge_roots(
                nodes[lhs].child[0], nodes[rhs].child[0], left, middle);
            nodes[lhs].child[1] = merge_roots(
                nodes[lhs].child[1], nodes[rhs].child[1], middle + 1, right);
        }
        return lhs;
    }

public:
    MergeableSegmentTree(int minimum_coordinate, int maximum_coordinate)
        : left_bound(minimum_coordinate), right_bound(maximum_coordinate) {
        assert(minimum_coordinate <= maximum_coordinate);
    }

    void add(int& root, int position, i64 delta) {
        assert(left_bound <= position && position <= right_bound);
        root = update(root, left_bound, right_bound, position, delta);
    }

    int merge(int lhs, int rhs) {
        return merge_roots(lhs, rhs, left_bound, right_bound);
    }

    i64 total(int root) const {
        return nodes[root].sum;
    }

    optional<int> kth(int root, i64 rank) const {
        if (rank <= 0 || rank > nodes[root].sum) return nullopt;
        int left = left_bound;
        int right = right_bound;
        while (left != right) {
            int middle = left + (right - left) / 2;
            int left_child = nodes[root].child[0];
            if (rank <= nodes[left_child].sum) {
                root = left_child;
                right = middle;
            } else {
                rank -= nodes[left_child].sum;
                root = nodes[root].child[1];
                left = middle + 1;
            }
        }
        return left;
    }
};
```

单点修改和第 $k$ 小为 $O(\log C)$。合并复杂度与两个树同时存在节点的位置数有关；一批树总合并通常可按被消去节点总数分析。使用 `kth` 时所有叶子权值必须非负。
