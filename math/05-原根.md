# 原根

若模 $m$ 的乘法群是循环群，则原根 $g$ 的阶等于 $\varphi(m)$。通过分解 $\varphi(m)$，只需检查每个不同质因子 $q$：

$$
g^{\varphi(m)/q}\not\equiv1\pmod m.
$$

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

using i64 = long long;
using i128 = __int128_t;

i64 power_mod(i64 base, i64 exponent, i64 modulus) {
    i64 result = 1 % modulus;
    while (exponent > 0) {
        if (exponent & 1) result = (i64)((i128)result * base % modulus);
        base = (i64)((i128)base * base % modulus);
        exponent >>= 1;
    }
    return result;
}

vector<i64> distinct_prime_factors(i64 value) {
    vector<i64> factors;
    for (i64 prime = 2; prime <= value / prime; ++prime) {
        if (value % prime != 0) continue;
        factors.push_back(prime);
        while (value % prime == 0) value /= prime;
    }
    if (value > 1) factors.push_back(value);
    return factors;
}

i64 euler_phi(i64 value) {
    i64 result = value;
    for (i64 prime : distinct_prime_factors(value)) {
        result = result / prime * (prime - 1);
    }
    return result;
}

i64 minimum_primitive_root(i64 modulus) {
    assert(modulus >= 2);
    if (modulus == 2) return 1;
    i64 phi = euler_phi(modulus);
    vector<i64> factors = distinct_prime_factors(phi);
    for (i64 candidate = 2; candidate < modulus; ++candidate) {
        if (gcd(candidate, modulus) != 1) continue;
        bool valid = true;
        for (i64 factor : factors) {
            if (power_mod(candidate, phi / factor, modulus) == 1) {
                valid = false;
                break;
            }
        }
        if (valid) return candidate;
    }
    return -1;
}
```

存在原根的正整数恰为 $2,4,p^k,2p^k$，其中 $p$ 是奇质数。函数在不存在时返回 -1；试除分解与枚举候选适合中等模数，64 位大模数应配合 Pollard–Rho。
