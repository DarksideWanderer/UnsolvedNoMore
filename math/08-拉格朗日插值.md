# 拉格朗日插值

给定模质数意义下互不相同的横坐标，计算次数小于点数的唯一多项式在 `target` 处的值。复杂度为 $O(n^2+n\log p)$。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

int lagrange_interpolation(const vector<int>& x, const vector<int>& y,
                           int target, int prime) {
    assert(x.size() == y.size());
    int size = (int)x.size();
    target %= prime;
    if (target < 0) target += prime;

    auto power_mod = [&](long long base, int exponent) {
        long long result = 1;
        while (exponent > 0) {
            if (exponent & 1) result = result * base % prime;
            base = base * base % prime;
            exponent >>= 1;
        }
        return (int)result;
    };

    long long answer = 0;
    for (int point = 0; point < size; ++point) {
        int normalized_x = (x[point] % prime + prime) % prime;
        if (normalized_x == target) return (y[point] % prime + prime) % prime;
        long long numerator = 1;
        long long denominator = 1;
        for (int other = 0; other < size; ++other) {
            if (other == point) continue;
            int other_x = (x[other] % prime + prime) % prime;
            numerator = numerator * (target - other_x + prime) % prime;
            denominator =
                denominator * (normalized_x - other_x + prime) % prime;
        }
        assert(denominator != 0);
        long long normalized_y = (y[point] % prime + prime) % prime;
        answer += normalized_y * numerator % prime *
                  power_mod(denominator, prime - 2) % prime;
        answer %= prime;
    }
    return (int)answer;
}
```

若横坐标恰为连续的 $0,1,\ldots,n-1$，可预处理阶乘、逆阶乘和目标点的前后缀积，将单次求值降为 $O(n)$。
