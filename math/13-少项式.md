# 少项式技巧

少项式指只有少量非零项的多项式。应把复杂度写成非零项数 $m$ 的函数，而不是最高次数的函数。

## 模意义下的函数迭代

设

$$
f(x)=\sum_{(d,a_d)}a_dx^d,\qquad
x_{t+1}=f(x_t)\bmod p.
$$

因为 $f(x)\bmod p$ 只依赖 $x\bmod p$，余数 $0,1,\ldots,p-1$ 构成一张函数图。先计算每个余数的一步转移，再对转移函数倍增，即可在 $O(\log y)$ 内求 $f^{(y)}(x)$。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

using i64 = long long;
using i128 = __int128_t;
using u64 = unsigned long long;

class SparsePolynomialIteration {
    int modulus;
    vector<vector<int>> jump;

    int power_mod(i64 base, u64 exponent) const {
        i64 result = 1 % modulus;
        base %= modulus;
        if (base < 0) base += modulus;
        while (exponent > 0) {
            if (exponent & 1) result = (i64)((i128)result * base % modulus);
            base = (i64)((i128)base * base % modulus);
            exponent >>= 1;
        }
        return (int)result;
    }

public:
    SparsePolynomialIteration(
        int iteration_modulus, const vector<pair<u64, i64>>& terms,
        u64 maximum_steps) : modulus(iteration_modulus) {
        assert(iteration_modulus >= 1);
        int level_count = max(1, (int)bit_width(maximum_steps));
        jump.assign(level_count, vector<int>(modulus));
        for (int value = 0; value < iteration_modulus; ++value) {
            i64 next = 0;
            for (auto [degree, coefficient] : terms) {
                coefficient %= modulus;
                if (coefficient < 0) coefficient += modulus;
                next = (i64)((next + (i128)coefficient *
                              power_mod(value, degree)) % modulus);
            }
            jump[0][value] = (int)next;
        }
        for (int level = 1; level < level_count; ++level) {
            for (int value = 0; value < modulus; ++value) {
                jump[level][value] =
                    jump[level - 1][jump[level - 1][value]];
            }
        }
    }

    int iterate(i64 start, u64 steps) const {
        int value = (int)(start % modulus);
        if (value < 0) value += modulus;
        for (int level = 0; steps > 0; ++level, steps >>= 1) {
            assert(level < (int)jump.size());
            if (steps & 1) value = jump[level][value];
        }
        return value;
    }
};
```

预处理一步转移需要 $O(pm\log D)$，其中 $D$ 是最大指数；倍增需要 $O(p\log y)$ 时间和空间。构造时的 `maximum_steps` 必须不小于之后任何询问的步数。若 $p$ 很大而查询很少，直接在函数图上做环检测通常更省空间。

## 少项式的形式幂级数幂

令 $G(x)=A(x)^k$。由

$$
G'(x)A(x)=kG(x)A'(x)
$$

比较 $x^n$ 的系数可得

$$
(n+1)a_0g_{n+1}
=\sum_{d=1}^{n+1}\bigl((k+1)d-(n+1)\bigr)a_dg_{n+1-d}.
$$

因此若 $A$ 只有 $m$ 个非零项，可以在 $O(Nm)$ 内求出 $A(x)^k\bmod x^{N+1}$。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

using i64 = long long;
using i128 = __int128_t;

vector<int> sparse_power_series(vector<pair<int, int>> terms,
                                i64 exponent, int degree,
                                int prime) {
    assert(exponent >= 0 && 0 <= degree && degree < prime);
    auto normalize = [&](i64 value) {
        value %= prime;
        if (value < 0) value += prime;
        return (int)value;
    };
    auto power_mod = [&](i64 base, i64 power) {
        i64 result = 1;
        while (power > 0) {
            if (power & 1) result = result * base % prime;
            base = base * base % prime;
            power >>= 1;
        }
        return (int)result;
    };

    map<int, int> merged;
    for (auto [term_degree, coefficient] : terms) {
        assert(term_degree >= 0);
        int& current = merged[term_degree];
        current = normalize((i64)current + coefficient);
    }
    vector<pair<int, int>> nonconstant_terms;
    int constant = merged.contains(0) ? merged[0] : 0;
    assert(constant != 0);
    for (auto [term_degree, coefficient] : merged) {
        if (term_degree > 0 && term_degree <= degree && coefficient != 0) {
            nonconstant_terms.push_back({term_degree, coefficient});
        }
    }

    vector<int> result(degree + 1);
    result[0] = power_mod(constant, exponent);
    i64 exponent_mod = exponent % prime;
    for (int current_degree = 1;
         current_degree <= degree; ++current_degree) {
        i64 numerator = 0;
        for (auto [term_degree, coefficient] : nonconstant_terms) {
            if (term_degree > current_degree) break;
            i64 factor = normalize(
                (i64)((i128)(exponent_mod + 1) * term_degree -
                      current_degree));
            numerator = (i64)((numerator + (i128)factor * coefficient % prime *
                               result[current_degree - term_degree]) % prime);
        }
        int denominator = (int)((i64)current_degree * constant % prime);
        result[current_degree] = (int)(
            numerator * power_mod(denominator, prime - 2) % prime);
    }
    return result;
}
```

这里要求 `prime` 为质数、$a_0\ne0$、$0\le N<prime$ 且 $k\ge0$，以保证递推中的分母可逆。输入中的同次项会先合并。若 $a_0=0$，应先提取最低次幂；通用的大规模多项式幂见 `16-polynomial.md`。
