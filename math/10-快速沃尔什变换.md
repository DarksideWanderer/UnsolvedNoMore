# 快速沃尔什变换

数组长度必须是 2 的幂，所有值位于模数范围内。OR/AND 逆变换只需加减；XOR 逆变换需要 2 在模意义下可逆。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

enum class BitwiseConvolution {
    bit_or,   // 下标按位或卷积
    bit_and,  // 下标按位与卷积
    bit_xor   // 下标按位异或卷积
};

void fwt(vector<int>& values, BitwiseConvolution type,
         bool inverse, int modulus) {
    int size = (int)values.size();
    assert(size > 0 && (size & (size - 1)) == 0);
    int inverse_two = (modulus + 1) / 2;
    for (int half = 1; half < size; half <<= 1) {
        for (int begin = 0; begin < size; begin += half * 2) {
            for (int offset = 0; offset < half; ++offset) {
                int lhs = values[begin + offset];
                int rhs = values[begin + half + offset];
                if (type == BitwiseConvolution::bit_or) {
                    values[begin + half + offset] = inverse
                        ? (rhs - lhs + modulus) % modulus
                        : (rhs + lhs) % modulus;
                } else if (type == BitwiseConvolution::bit_and) {
                    values[begin + offset] = inverse
                        ? (lhs - rhs + modulus) % modulus
                        : (lhs + rhs) % modulus;
                } else {
                    values[begin + offset] = (lhs + rhs) % modulus;
                    values[begin + half + offset] =
                        (lhs - rhs + modulus) % modulus;
                    if (inverse) {
                        values[begin + offset] = (int)(
                            (long long)values[begin + offset] *
                            inverse_two % modulus);
                        values[begin + half + offset] = (int)(
                            (long long)values[begin + half + offset] *
                            inverse_two % modulus);
                    }
                }
            }
        }
    }
}

vector<int> bitwise_convolution(vector<int> lhs, vector<int> rhs,
                                BitwiseConvolution type, int modulus) {
    assert(lhs.size() == rhs.size());
    fwt(lhs, type, false, modulus);
    fwt(rhs, type, false, modulus);
    for (int i = 0; i < (int)lhs.size(); ++i) {
        lhs[i] = (int)((long long)lhs[i] * rhs[i] % modulus);
    }
    fwt(lhs, type, true, modulus);
    return lhs;
}
```

复杂度为 $O(n2^n)$（数组长度为 $2^n$）。XOR 版本要求模数为奇数；常用质数模数满足这一点。
