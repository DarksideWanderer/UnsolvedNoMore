# 子集卷积

计算

$$
h(S)=\sum_{T\subseteq S}f(T)g(S\setminus T).
$$

使用按集合大小分层的子集 Zeta 变换，复杂度为 $O(n^2 2^n)$，空间复杂度为 $O(n2^n)$。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

void subset_zeta(vector<int>& values, int bit_count,
                 bool inverse, int modulus) {
    int size = 1 << bit_count;
    for (int bit = 0; bit < bit_count; ++bit) {
        for (int mask = 0; mask < size; ++mask) {
            if (!(mask >> bit & 1)) continue;
            if (inverse) {
                values[mask] -= values[mask ^ (1 << bit)];
                if (values[mask] < 0) values[mask] += modulus;
            } else {
                values[mask] += values[mask ^ (1 << bit)];
                if (values[mask] >= modulus) values[mask] -= modulus;
            }
        }
    }
}

vector<int> subset_convolution(const vector<int>& lhs,
                               const vector<int>& rhs,
                               int modulus) {
    assert(lhs.size() == rhs.size());
    int size = (int)lhs.size();
    assert(size > 0 && (size & (size - 1)) == 0);
    int bit_count = countr_zero((unsigned)size);

    vector<vector<int>> lhs_layer(bit_count + 1, vector<int>(size));
    vector<vector<int>> rhs_layer(bit_count + 1, vector<int>(size));
    vector<vector<int>> result_layer(bit_count + 1, vector<int>(size));
    for (int mask = 0; mask < size; ++mask) {
        int bits = popcount((unsigned)mask);
        lhs_layer[bits][mask] = lhs[mask];
        rhs_layer[bits][mask] = rhs[mask];
    }
    for (int bits = 0; bits <= bit_count; ++bits) {
        subset_zeta(lhs_layer[bits], bit_count, false, modulus);
        subset_zeta(rhs_layer[bits], bit_count, false, modulus);
    }
    for (int total_bits = 0; total_bits <= bit_count; ++total_bits) {
        for (int lhs_bits = 0; lhs_bits <= total_bits; ++lhs_bits) {
            for (int mask = 0; mask < size; ++mask) {
                result_layer[total_bits][mask] = (int)(
                    (result_layer[total_bits][mask] +
                     (long long)lhs_layer[lhs_bits][mask] *
                     rhs_layer[total_bits - lhs_bits][mask]) % modulus);
            }
        }
        subset_zeta(result_layer[total_bits], bit_count, true, modulus);
    }

    vector<int> result(size);
    for (int mask = 0; mask < size; ++mask) {
        result[mask] = result_layer[popcount((unsigned)mask)][mask];
    }
    return result;
}
```

输入系数必须先正规化到 $[0,modulus)$。当 $n=21$ 时三组二维数组已经很大，应按内存限制选择静态数组、滚动层或其他题目特定优化。
