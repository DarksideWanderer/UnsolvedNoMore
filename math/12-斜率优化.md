# 斜率优化

```cpp
void Main(int Case){
	int n,m,p;InPut(n,m,p);
	vector<int>d(n+1);
	rep(i,2,n){
		InPut(d[i]);
		d[i]+=d[i-1];
	}
	vector<int>Cr(m+1);
	for(int i=1;i<=m;i++){
		int x,y;InPut(x,y);
		Cr[i]=y-d[x];
	}
	std::sort(wqlall(Cr));
	vector<int>Sum(m+1);
	Sum[0]=0;
	for(int i=1;i<=m;i++){
		Sum[i]=Sum[i-1]+Cr[i];
	}
	vector<vector<int>>dp(m+1,vector<int>(p+1,inf));//收走了i个点,用掉为j个人的方案花费最小为dp[i][j]
	dp[0][0]=0;//初始状态,没有点,没有人,花费为0
	static auto X=[&](int x)->int {
		return x;
	};
	static auto Y=[&](int x,int j)->int {
		return dp[x][j-1]+Sum[x];
	};
	rep(j,1,p){
		std::deque<int>dq;
		dp[0][j]=0;
		dq.push_back(0);
		rep(i,1,m){
			while(sz(dq)>1&&(I*(Y(dq[0],j)-Y(dq[1],j)))>=(I*Cr[i]*(X(dq[0])-X(dq[1])))){
				dq.pop_front();
			}
			int t=dq.front();
			dp[i][j]=dp[t][j-1]+Sum[t]-Sum[i]+i*Cr[i]-t*Cr[i];
			if(i==m)break;
			while(sz(dq)>1&&I*(Y(i,j)-Y(dq[sz(dq)-1],j))*(X(dq.back())-X(dq[sz(dq)-2]))<=
							I*(Y(dq.back(),j)-Y(dq[sz(dq)-2],j))*(X(i)-X(dq[sz(dq)-1]))){
				dq.pop_back();
			}
			dq.push_back(i);
		}
	}
	OutPut(dp[m][p],'\n');
}
```
