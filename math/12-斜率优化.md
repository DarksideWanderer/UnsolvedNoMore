# 单调斜率优化

维护直线 $y=kx+b$ 的最小值。要求斜率按严格递减顺序加入、查询横坐标单调不减；相同斜率只保留截距更小者。比较使用 `__int128_t`，前提是斜率、截距、横坐标的乘积不超过 128 位范围。

```cpp
#include <bits/stdc++.h>
#include <cassert>
using namespace std;

class MonotoneConvexHull {
    using i64 = long long;
    using i128 = __int128_t;

    struct Line {
        i64 slope;
        i64 intercept;

        i128 value(i64 x) const {
            return (i128)slope * x + intercept;
        }
    };

    deque<Line> lines;

    static bool redundant(const Line& lhs, const Line& middle,
                          const Line& rhs) {
        return ((i128)middle.intercept - lhs.intercept) *
                   ((i128)middle.slope - rhs.slope) >=
               ((i128)rhs.intercept - middle.intercept) *
                   ((i128)lhs.slope - middle.slope);
    }

public:
    void add_line(i64 slope, i64 intercept) {
        if (!lines.empty() && lines.back().slope == slope) {
            if (lines.back().intercept <= intercept) return;
            lines.pop_back();
        }
        assert(lines.empty() || lines.back().slope > slope);
        Line added{slope, intercept};
        while (lines.size() >= 2 &&
               redundant(lines[lines.size() - 2], lines.back(), added)) {
            lines.pop_back();
        }
        lines.push_back(added);
    }

    i128 query(i64 x) {
        assert(!lines.empty());
        while (lines.size() >= 2 &&
               lines[1].value(x) <= lines[0].value(x)) {
            lines.pop_front();
        }
        return lines.front().value(x);
    }
};
```

每条直线最多入队、出队一次，因此一批操作总复杂度为线性。若斜率或查询横坐标不单调，使用李超线段树；若只离线查询，可按横坐标排序后恢复原顺序。
